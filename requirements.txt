# Qwen-Image-2.1 (Uncensored Omni, FP8) ワーカー用
# torch/torchvision は必ずペアで同じ CUDA ビルドにすること。
# （片方だけ違うと torchvision::nms エラーでワーカーが起動時に落ちる）
# GPU は NVIDIA RTX PRO 6000 Blackwell (sm_120) のため cu128 ビルドが必要。
# torch は 2.10.0 が必要：torchao の float8 モジュールが torch.nn.functional の
# ScalingType を import するが、これは torch 2.10 で追加されたもので 2.8.0 には
# 無い。torch 2.8.0 のままだとワーカーが起動時の import で落ちる
# （2026-11-26 の RunPod ログで確認：cannot import name 'ScalingType'）。
--extra-index-url https://download.pytorch.org/whl/cu128
torch==2.10.0+cu128
torchvision==0.25.0+cu128
# torchaudio も torch と同じバージョンに固定する。固定しないとベースイメージの
# 古い torchaudio（2.4 系）が残り、torch 2.10 と組み合わさって libtorchaudio.so
# の読み込みに失敗し、ワーカーが起動時の import で落ちる
# （2026-11-26 の RunPod ログで確認：Could not load this library: libtorchaudio.so）。
torchaudio==2.10.0+cu128
# diffusers 0.41.0 以降でないと QwenImage21Pipeline が無い（0.40.0 には無い）。
diffusers==0.41.0
# transformers は 5.x が必須。4.x（最新 4.57.6）は huggingface-hub<1.0 を
# 要求するが、diffusers 0.41.0 は huggingface-hub>=1.32.0 を要求するため、
# 両方を満たすバージョンが存在せずビルドが ResolutionImpossible で失敗する
# （2026-11-25 の初回ビルドで確認）。5.x は huggingface-hub>=1.31.0 を
# 要求し、diffusers 0.41.0 と重なる。
transformers==5.24.1
accelerate==0.34.2
# FP8-torchao の transformer を読むのに必要。
# torchao は torch 2.10 とペアの 0.15.0 に固定する。固定しないと新しい
# torchao（0.18.0 など）が入り、古い torch では import 時に落ちる。
torchao==0.15.0
runpod
safetensors
Pillow
sentencepiece
protobuf
qwen-vl-utils
